## 1. 手写冒泡排序
```javascript
// bubbleSort.js
const bubbleSort = (arr) => {
  const n = arr.length;
  for (let i = 0; i < n - 1; i++) {
    for (let j = 0; j < n - 1 - i; j++) {
      if (arr[j] > arr[j + 1]) {
        [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]];
      }
    }
  }
  return arr;
}
// 运行测试
const nums = [5, 3, 8, 4, 2];
console.log(bubbleSort(nums));
```

## 2. 手写new关键字源码
```javascript
// new.js

function Student(name, age) {
    this.name = name;
    this.age = age;
}

function _new() {
    let obj={};
    // 取出所有形参
    let params = Array.prototype.slice.call(arguments, 1);
    // 取出构造函数
    let fn = arguments[0];
    // obj的原型指向fn的原型
    obj.__proto__ = fn.prototype;
    // obj的原型的构造函数指向fn
    obj.__proto__.constructor = fn;
    //修改this指向,让obj运行按参数来运行构造函数
    fn.apply(obj, params);
    return obj;
}

// 运行测试
```


## 3. 手写instanceof源码
```javascript
// instanceof.js
function _instanceof(child, parent) {
    var left = child.__proto__;
    var right = parent.prototype;

    while(true) {
        if (left === null) {
            return false;
        }
        if (left === right) {
            return true;
        }
        // 如果left不等于right，则用left的父集和right比，如果还不相等则用left的父集的父集，以此在原型链上向上找
        left = left.__proto__;
    }
}

// 运行测试
console.log(_instanceof(new Date(), Array));
console.log(_instanceof(new Date(), Date));
console.log(_instanceof([], Object));
```
主机上运行，返回结果如下
```bash
$ node instanceof.js 
false
true
true
```

## 4. 手写防抖和节流
### 4.1 防抖
```javascript
// debounce.js

// 原理：将业务封装到一个定时器里
// 业务频繁调用，每次会清除前一个定时器，只保证有1个定时器当前在运行
// 从而避免了在实际情况下多次调用业务
function debounce(fn){
    var timer=null; //使用闭包，防止全局污染
    return function(){
        clearTimeout(timer); // 清除之前的timer
        // 创建运行新的timer
        timer = setTimeout(function() {
            fn();
        }, 1000);
    }
}
// 测试用函数
const fn = () => {
    console.log("click");
}
// 绑定onclick
btn.onclick=debounce(fn);
```
在某个button的onclick事件上绑定该防抖函数，使用浏览器打开如下html页面，点击按钮之后在控制台上可以观察到防抖现象

### 4.2 节流
```javascript
// throttle.js

// 原理：将业务封装到定时器里面
// 通过设置flag，只有当定时器执行完毕之后才重置flag
// 在flag重置之前，所有重复的业务调用都不会执行
const throttle = (fn) => {
    var flag = true;
    return function(){
        if(!flag) {
            return
        } else {
            flag = false;
            setTimeout(function(){
                fn();
                flag = true;
            },1000);
        }
    }
}
// 测试用函数
const fn = () => {
    console.log("run");
}
// 绑定onclick
btn.onclick = throttle(fn);
```
在某个button的onclick事件上绑定该防抖函数，使用浏览器打开如下html页面，点击按钮之后在控制台上可以观察到节流现象

## 5. typeof运算符的结果一共有哪些可能，并说出与instanceof的区别
```text
typeof返回 boolean, number, string, object, function, undefined, symbol这些类型

和instanceof的不同
```

## 6. 一句话解释什么是原型链
```text
一个对象的隐式原型指向父集的原型，而父集也有隐式原型，也会指向他的父集原型，...，按这种指向一直向上指，直到指向null为止。这种层层指向的关系，构成原型链
```

## 9. 跨域问题是如何产生的
```text
当浏览器请求不属于本站的数据的时候，比如本地的http://localhost 里面的js代码要请求 http://example.com 的数据时，可能会产生跨域问题
过程：从浏览器软件发出的HTTP请求，当HTTP请求的Host头和Origin头不一致时（如上例, Origin: localhost, Host: example.com），服务端接收到该HTTP请求经过处理之后，返回给浏览器软件HTTP的响应里面如果没有将当时Origin头的相关信息添加到响应头Access-Control-Allow-Origin时，造成的浏览器软件单方面丢掉响应数据的情况，称为跨域问题
```

## 12. call，apply，bind 三者用法和区别
```text
这3个都是用来改变this指向的
obj2.fun.call(obj1, arg1, arg2, ...) // obj1调用obj2的fun方法，这时obj2里面的this指向obj1. arg是fun需要的参数

obj2.fun.apply(obj1, [arg1, arg2, arg3, ...]) // obj1调用obj2的fun方法，这时obj2里面的this指向obj1, arg是fun需要的参数，和call不同的是apply传的参数是数组形式的

obj2.fun.bind(obj1, arg1, arg2, ...) // 返回填了arg的fun函数，这个函数是obj1调用obj2的fun函数，里面的this指向obj1, 和上面2个不同的是bind只返回函数并不执行这个函数需要人工执行
```

## 13. 至少用三种方式实现数组去重

```javascript
// unique.js
const origin = [1,1,1,2,2,2,3,3,3];


// 1. 使用Set + ...解构方式
const unique1 = (arr) => [...new Set(arr)];
console.log(unique1(origin));


// 2. 使用filter + Map来筛
const unique2 = (arr) => {
    const map = new Map();
    return arr.filter(item => !map.has(item) && map.set(item, true));
}

// 3. for循环遍历
const unique3 = (arr) => {
    const bucket1 = {};
    const bucket2 = [];
    for (let i = 0; i < arr.length; i++) {
        const val = arr[i];
        if (!bucket1[val]) {
            bucket1[val] = true;
            bucket2.push(val);
        }

}
```

## 16. 下面代码输出什么
```javascript
for (var i = 0; i < 3; i++) {
    setTimeout(()=>console.log(i), 1);
}
for (let i = 0; i < 3; i++) {
    setTimeout(()=>console.log(i), 1);
}
```
var是全局作用域，由于setTimeout是异步的，当孵化的3个异步线程执行之前，for循环已经走完了，即在3个异步线程执行之前，i已经变成了3，因此执行结果是3 3 3

let是块级作用域，每轮孵化一个异步线程，都会创建一个新的i，每一个setTimeout的闭包都记录了当时的i，因此执行结果是0 1 2

## 19. 简单描述什么是事件循环
```text
js的一种运行机制，用于处理异步操作. 事件循环通过维护任务队列来处理异步操作. 任务队列分为2个，宏任务队列 和 微任务队列。宏任务队列存储整体的js,setTimeout,setInterval. 微任务队列存储Promise对象的then()方法里面的内容.每轮宏人物执行的时候，必须清空当轮微任务，才能进行下一个宏任务. 

事件循环的执行顺序
1. 执行当前宏人物
2. 检查微任务队列是否清空，如果没有清空，那么按照条件顺序依次执行，直到任务列表被清空
3. 从宏任务队列中取出下一个宏任务，执行它(2, 3循环)
```

```javascript
var h1 = document.getElementById("txt");
var xmlhttp = new XMLHttpRequest();
xmlhttp.open("get", "http://localhost:8888/api/hello", true);
xmlhttp.send(null);

        h1.innerText = data;
    }

// post请求
var like = document.getElementById("like");
var h2 = document.getElementById("txt2");
like.onblur = () => {
    var xmlhttp = new XMLHttpRequest();
    xmlhttp.open("post", "http://localhost:8888/api/hobby", true);
    // 设置请求头为x-www-form-urlencoded
    xmlhttp.setRequestHeader("Content-Type","application/x-www-form-urlencoded");
    xmlhttp.send("hobby=" + like.value);
    xmlhttp.onreadystatechange = () => {
        if (xmlhttp.readyState === 4 && (xmlhttp.status >= 200 && xmlhttp.status < 300)) {
            var data = xmlhttp.responseText;
            h2.innerText = data;
        }
    }
```
